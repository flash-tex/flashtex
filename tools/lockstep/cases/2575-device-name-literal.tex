\closein0
\openin0=TeXfonts:/etc/hosts
\ifeof0\message{2575 fonts-area eof}\else\message{2575 fonts-area read}\fi
\closein0
\immediate\openout1=TeXinputs:ls2575.tex
\immediate\write1{\relax}
\immediate\closeout1
\openin0=ls2575.tex
\ifeof0\message{2575 stripped-name eof}\else\message{2575 stripped-name found}\fi
\closein0
