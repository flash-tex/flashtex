\immediate\openout1=ls1271a.txt \immediate\write1{ALPHA} \immediate\closeout1
\immediate\openout1=ls1271b.txt \immediate\write1{BETA} \immediate\closeout1
\openin0=ls1271a.txt
\openin0=ls1271b.txt
\read0 to \gotB
\setbox0=\hbox{\myrm\gotB}
\openin0=ls1271a.txt
\read0 to \gotA \closein0
\setbox1=\hbox{\myrm\gotA}
\lsshipbox0 \lsshipbox1
