\def\nextA{\ifx\peeked\spacetoken \count0=1 \message{716 got-space}\else\message{716 got-other}\fi}

\message{716 peek-mean=[\meaning\peeked]}
\setbox0=\hbox{\vrule width716pt height8pt depth2pt\hskip\count0 pt}
\count0=0

\futurelet\peeked\nextB\fooA
