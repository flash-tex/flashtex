% 2101-expand-ratio-zero-glueset: one-line paragraphs set a few sp off their natural width under \pdfadjustspacing=2, so post_line_break's hpack(q,cur_width,cal_expand_ratio) gets font_expand_ratio=0 and returns before setting glue_set/glue_sign/glue_order, which keep what get_node left there, so the dump depends on SyncTeX's node sizes (pdftex.web §§823, 834, 840, 1066)
\input prelude
\font\myrm=cmr10
\pdffontexpand\myrm 30 20 10 autoexpand
\parindent=0pt \parfillskip=0pt \pdfadjustspacing=2
\countdef\k=10
\def\txt{This is a small test line of text.}
\setbox1=\hbox{\myrm\txt}
\def\fA{{\setbox9=\hbox to 50pt{\hfil}}}
\def\fB{{\setbox9=\hbox to 50pt{\hskip 0pt plus 1pt}}}
\def\fC{{\setbox9=\hbox to 5pt{\hskip 0pt minus 10pt x}}}
\def\fD{{\setbox9=\vbox to 50pt{\vskip 0pt plus 1fil}}}
\def\one{\hsize=\wd1 \advance\hsize by \k sp
  \myrm\fA\txt\fA\par
  \myrm\txt\fB\par
  \myrm\fC\txt\fC\par
  \myrm\fD\txt\fA\fB\fC\fD\par}
\def\step{\ifnum\k<12 \advance\k by 1 \ifnum\k=0 \else\one\fi\expandafter\step\fi}
\setbox0=\vbox{%
  \hsize=100pt \myrm This is a longer paragraph that breaks into several lines so that the line breaker allocates and frees active, passive and delta nodes before the short lines. \par
  \k=-13 \step}
\lsshipbox0
\end
