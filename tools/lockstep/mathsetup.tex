% tools/lockstep/mathsetup.tex -- shared plain-style math fonts for lockstep cases.
% -ini mode loads no format, so no math fonts exist yet: any math formula
% dies with `! Math formula deleted: Insufficient symbol fonts.' unless
% \textfont2/\scriptfont2/\scriptscriptfont2 have >= 22 \fontdimen
% parameters (cmsy10/cmsy7/cmsy5) and \textfont3/\scriptfont3/
% \scriptscriptfont3 have >= 13 (cmex10). This mirrors plain.tex exactly
% (\font\tenrm=cmr10 etc., skewchars, families 0-3 assignments).
% NOTE: cases currently inline this block verbatim instead of
% `\input mathsetup' because run.py stages only prelude.tex plus the case
% file into the run dir, so a bare `\input mathsetup' cannot resolve
% (emergency stop, exit 1 -- verified). If the harness ever stages shared
% includes, replace each inline block with `\input mathsetup' right after
% `\input prelude'.
\font\tenrm=cmr10 \font\sevenrm=cmr7 \font\fiverm=cmr5
\font\teni=cmmi10 \font\seveni=cmmi7 \font\fivei=cmmi5
\font\tensy=cmsy10 \font\sevensy=cmsy7 \font\fivesy=cmsy5
\font\tenex=cmex10
\skewchar\teni='177 \skewchar\seveni='177 \skewchar\fivei='177
\skewchar\tensy='60 \skewchar\sevensy='60 \skewchar\fivesy='60
\textfont0=\tenrm \scriptfont0=\sevenrm \scriptscriptfont0=\fiverm
\textfont1=\teni \scriptfont1=\seveni \scriptscriptfont1=\fivei
\textfont2=\tensy \scriptfont2=\sevensy \scriptscriptfont2=\fivesy
\textfont3=\tenex \scriptfont3=\tenex \scriptscriptfont3=\tenex
