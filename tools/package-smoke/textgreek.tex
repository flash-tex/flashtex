\documentclass{article}
\usepackage{textgreek}
\begin{document}
This package prints Greek letters in ordinary text mode.
Lowercase: \textalpha, \textbeta, \textgamma, \textdelta, and \textomega\ appear.
Capitals: \textAlpha, \textBeta, \textGamma, and \textOmega\ appear.
Mixed sentence: The \textmu\ particle moves at \textnu\ hertz here.
\end{document}
