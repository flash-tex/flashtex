\begin{algorithm}[h]
\caption{Wrapping}
\begin{algorithmic}[1]
\FOR{each vertex}
\IF{the vertex has not been visited yet}
\STATE mark the vertex as visited and push every one of its neighbours onto the stack so that they are explored later in depth-first order
\ENDIF
\ENDFOR
\end{algorithmic}
\end{algorithm}
