\begin{algorithm}[h]
\caption{Nesting}
\begin{algorithmic}[1]
\WHILE{\TRUE}
\FOR{$i \gets 1$ \TO $n$}
\IF{$a_i > m$}
\STATE $m \gets a_i$
\ELSIF{$a_i = m$}
\STATE count it
\ENDIF
\ENDFOR
\ENDWHILE
\end{algorithmic}
\end{algorithm}
