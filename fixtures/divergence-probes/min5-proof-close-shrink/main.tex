\begin{theorem}[Division with remainder]
Let $a \in \mathbb{Z}$ and $b > 0$. There exist unique $q, r$ with
\[
  a = qb + r \quad\text{and}\quad 0 \le r < b.
\]
\end{theorem}

\begin{proof}
\emph{Existence.} Take the least non-negative element of the set of
values $a - kb$; it exists by well-ordering.

\emph{Uniqueness.} Two such remainders differ by a multiple of $b$ that
is smaller than $b$, hence by zero.
\end{proof}

\begin{corollary}
Every integer is either even or odd, and not both.
\end{corollary}
