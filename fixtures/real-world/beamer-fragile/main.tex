\documentclass{beamer}
% Frame options: [fragile] with verbatim and lstlisting, [allowframebreaks]
% with a list that spans two pages, the \frame{...} command form, [plain],
% and \note{...} which must not appear in the output.
\usepackage{listings}
\lstset{basicstyle=\ttfamily\small}
\setbeamertemplate{navigation symbols}{}

\title{Fragile Frames}
\author{J. Whitfield}
\date{March 2026}

\begin{document}

\begin{frame}
  \titlepage
\end{frame}

\begin{frame}[fragile]
  \frametitle{Verbatim}
  A fragile frame may contain verbatim material.
\begin{verbatim}
\begin{frame}[fragile]
  \frametitle{Verbatim}   % needs 'fragile'
  ...
\end{verbatim}
  The lines above are typeset exactly as written.
\end{frame}

\begin{frame}[fragile]
  \frametitle{A listing}
\begin{lstlisting}
fn main() {
    let pages = render(&document);
    println!("{} pages", pages.len());
}
\end{lstlisting}
  \note{This note is for the presenter and must not be printed.}
\end{frame}

  \frametitle{The command form}

}

\begin{frame}[plain]
  \frametitle{Plain}
  A plain frame has no headline, footline or sidebars, and the text width
  is the full paper width minus the margins.
\end{frame}
